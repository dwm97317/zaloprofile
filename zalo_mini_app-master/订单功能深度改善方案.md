```php
public function advancedSearch() {
    $params = $this->request->param();
    
    $query = PackageModel::with(['country', 'storage', 'packageimage.file']);
    
    // 日期范围筛选
    if (!empty($params['date_start']) && !empty($params['date_end'])) {
        $query->whereBetweenTime('created_time', $params['date_start'], $params['date_end']);
    }
    
    // 价格区间筛选
    if (isset($params['price_min']) && isset($params['price_max'])) {
        $query->whereBetween('price', [$params['price_min'], $params['price_max']]);
    }
    
    // 多仓库筛选
    if (!empty($params['warehouse_ids'])) {
        $query->whereIn('storage_id', explode(',', $params['warehouse_ids']));
    }
    
    // 多国家筛选
    if (!empty($params['country_ids'])) {
        $query->whereIn('country_id', explode(',', $params['country_ids']));
    }
    
    // 重量范围筛选
    if (isset($params['weight_min']) && isset($params['weight_max'])) {
        $query->whereBetween('weight', [$params['weight_min'], $params['weight_max']]);
    }
    
    // 问题订单筛选
    if (!empty($params['has_issue'])) {
        $query->where('status', -1);
    }
    
    // 排序
    $sortBy = $params['sort_by'] ?? 'created_time';
    $sortOrder = $params['sort_order'] ?? 'desc';
    $query->order($sortBy, $sortOrder);
    
    // 分页
    $pageSize = $params['page_size'] ?? 20;
    $result = $query->paginate($pageSize);
    
    return $this->renderSuccess($result);
}
```

```php
// Package.php 控制器
public function batchOperation() {
    $action = $this->request->param('action');
    $ids = $this->request->param('ids');
    $params = $this->request->param('params', []);
    
    if (empty($ids) || !is_array($ids)) {
        return $this->renderError('请选择要操作的订单');
    }
    
    // 开启事务
    Db::startTrans();
    
    try {
        $success = 0;
        $failed = 0;
        $errors = [];
        
        foreach ($ids as $id) {
            try {
                switch ($action) {
                    case 'cancel':
                        $this->cancelPackage($id);
                        break;
                    case 'export':
                        $this->exportPackage($id);
                        break;
                    case 'tag':
                        $this->addTag($id, $params['tag']);
                        break;
                    case 'warehouse':
                        $this->transferWarehouse($id, $params['warehouse_id']);
                        break;
                    case 'priority':
                        $this->setPriority($id, $params['priority']);
                        break;
                    default:
                        throw new Exception('不支持的操作');
                }
                $success++;
            } catch (Exception $e) {
                $failed++;
                $errors[] = ['id' => $id, 'error' => $e->getMessage()];
            }
        }
        
        Db::commit();
        
        return $this->renderSuccess([
            'success' => $success,
            'failed' => $failed,
            'errors' => $errors
        ]);
        
    } catch (Exception $e) {
        Db::rollback();
        return $this->renderError('批量操作失败：' . $e->getMessage());
    }
}
```

#### 2.2.3 智能缓存策略

**功能描述**: 实现多层级缓存机制

**实现方案**:

```javascript
// 缓存管理器
class CacheManager {
  constructor() {
    this.memoryCache = new Map();
    this.storageCache = window.localStorage;
    this.cacheConfig = {
      orderList: { ttl: 5 * 60 * 1000, storage: 'memory' }, // 5分钟
      orderDetail: { ttl: 10 * 60 * 1000, storage: 'local' }, // 10分钟
      statistics: { ttl: 15 * 60 * 1000, storage: 'local' }, // 15分钟
      userInfo: { ttl: 60 * 60 * 1000, storage: 'local' } // 1小时
    };
  }
  
  set(key, value, type = 'orderList') {
    const config = this.cacheConfig[type];
    const cacheData = {
      value,
      timestamp: Date.now(),
      ttl: config.ttl
    };
    
    if (config.storage === 'memory') {
      this.memoryCache.set(key, cacheData);
    } else {
      this.storageCache.setItem(key, JSON.stringify(cacheData));
    }
  }
  
  get(key, type = 'orderList') {
    const config = this.cacheConfig[type];
    let cacheData;
    
    if (config.storage === 'memory') {
      cacheData = this.memoryCache.get(key);
    } else {
      const stored = this.storageCache.getItem(key);
      cacheData = stored ? JSON.parse(stored) : null;
    }
    
    if (!cacheData) return null;
    
    // 检查是否过期
    if (Date.now() - cacheData.timestamp > cacheData.ttl) {
      this.delete(key, type);
      return null;
    }
    
    return cacheData.value;
  }
  
  delete(key, type = 'orderList') {
    const config = this.cacheConfig[type];
    
    if (config.storage === 'memory') {
      this.memoryCache.delete(key);
    } else {
      this.storageCache.removeItem(key);
    }
  }
  
  clear(type) {
    if (type) {
      // 清除特定类型的缓存
      const config = this.cacheConfig[type];
      if (config.storage === 'memory') {
        this.memoryCache.clear();
      } else {
        // 清除localStorage中特定前缀的项
        Object.keys(this.storageCache)
          .filter(key => key.startsWith(type))
          .forEach(key => this.storageCache.removeItem(key));
      }
    } else {
      // 清除所有缓存
      this.memoryCache.clear();
      this.storageCache.clear();
    }
  }
}
```

```php
private function getTimeRange($range) {
    $now = time();
    switch ($range) {
        case 'today':
            return [
                'start' => strtotime('today'),
                'end' => $now
            ];
        case 'week':
            return [
                'start' => strtotime('this week'),
                'end' => $now
            ];
        case 'month':
            return [
                'start' => strtotime('first day of this month'),
                'end' => $now
            ];
        case 'year':
            return [
                'start' => strtotime('first day of January this year'),
                'end' => $now
            ];
        default:
            return [
                'start' => strtotime('today'),
                'end' => $now
            ];
    }
}

private function calculateAvgProcessTime($userId, $timeRange) {
    $completedOrders = Inpack::where('member_id', $userId)
        ->where('status', 8)
        ->whereBetweenTime('created_time', $timeRange['start'], $timeRange['end'])
        ->field('created_time, updated_time')
        ->select();
    
    if (empty($completedOrders)) return 0;
    
    $totalTime = 0;
    foreach ($completedOrders as $order) {
        $totalTime += strtotime($order['updated_time']) - strtotime($order['created_time']);
    }
    
    return round($totalTime / count($completedOrders) / 3600, 2); // 转换为小时
}
```

```javascript
class WebSocketManager {
  constructor(userId) {
    this.userId = userId;
    this.ws = null;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = 5;
    this.listeners = new Map();
  }
  
  connect() {
    const wsUrl = `wss://api.example.com/ws?user_id=${this.userId}`;
    this.ws = new WebSocket(wsUrl);
    
    this.ws.onopen = () => {
      console.log('WebSocket connected');
      this.reconnectAttempts = 0;
      this.sendHeartbeat();
    };
    
    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this.handleMessage(data);
    };
    
    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
    
    this.ws.onclose = () => {
      console.log('WebSocket closed');
      this.reconnect();
    };
  }
  
  handleMessage(data) {
    const { type, payload } = data;
    
    switch (type) {
      case 'order_status_changed':
        this.notifyListeners('statusChange', payload);
        this.showNotification(payload);
        break;
      case 'new_order':
        this.notifyListeners('newOrder', payload);
        break;
      case 'alert':
        this.notifyListeners('alert', payload);
        break;
    }
  }
  
  subscribe(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }
  
  notifyListeners(event, data) {
    const callbacks = this.listeners.get(event) || [];
    callbacks.forEach(callback => callback(data));
  }
  
  showNotification(data) {
    if ('Notification' in window && Notification.permission === 'granted') {
      new Notification('订单状态更新', {
        body: `订单 ${data.orderSn} 状态已更新为 ${data.statusText}`,
        icon: '/icon.png'
      });
    }
  }
  
  sendHeartbeat() {
    setInterval(() => {
      if (this.ws && this.ws.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ type: 'heartbeat' }));
      }
    }, 30000); // 每30秒发送一次心跳
  }
  
  reconnect() {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      const delay = Math.min(1000 * Math.pow(2, this.reconnectAttempts), 30000);
      setTimeout(() => this.connect(), delay);
    }
  }
  
  disconnect() {
    if (this.ws) {
      this.ws.close();
    }
  }
}
```

```javascript
    const deltaY = touch.clientY - this.touchStart.y;

      } else {
        this.onSwipeLeft && this.onSwipeLeft(e);

        this.onSwipeUp && this.onSwipeUp(e);
```

```javascript
    return () => {
      if (observerRef.current) {
        observerRef.current.disconnect();
      }
    };
  }, [hasMore, loading]);
  
  const loadMore = async () => {
    setLoading(true);
    await onLoadMore();
    setLoading(false);
  };
  
  return (
    <div className="infinite-scroll-container">
      {children}
      
      <div ref={sentinelRef} className="load-more-sentinel">
        {loading && (
          <div className="loading-indicator">
            <div className="spinner" />
            <span>加载中...</span>
          </div>
        )}
        
        {!hasMore && (
          <div className="no-more-data">
            没有更多数据了
          </div>
        )}
      </div>
    </div>
  );
};
```

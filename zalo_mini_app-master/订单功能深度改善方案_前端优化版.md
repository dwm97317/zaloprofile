```javascript
import { useVirtualizer } from '@tanstack/react-virtual';
import { useRef } from 'react';

const VirtualizedOrderList = ({ orders }) => {

          width: '100%',
          position: 'relative'

            <div

              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                width: '100%',
                height: `${virtualItem.size}px`,
                transform: `translateY(${virtualItem.start}px)`
              }}
            >
              <OrderCard order={order} />
            </div>
          );
        })}
      </div>
    </div>
  );
```

#### 2.1.3 增强缓存策略

**功能描述**: 实现多层级前端缓存机制

**实现方案**:

```javascript
// 缓存管理器
class CacheManager {
  constructor() {
    this.memoryCache = new Map();
    this.storageCache = window.localStorage;
    this.cacheConfig = {
      orderList: { ttl: 5 * 60 * 1000, storage: 'memory' }, // 5分钟
      orderDetail: { ttl: 10 * 60 * 1000, storage: 'local' }, // 10分钟
      statistics: { ttl: 15 * 60 * 1000, storage: 'local' }, // 15分钟
      userInfo: { ttl: 60 * 60 * 1000, storage: 'local' } // 1小时
    };
  }
  
  set(key, value, type = 'orderList') {
    const config = this.cacheConfig[type];
    const cacheData = {
      value,
      timestamp: Date.now(),
      ttl: config.ttl
    };
    
    if (config.storage === 'memory') {
      this.memoryCache.set(key, cacheData);
    } else {
      try {
        this.storageCache.setItem(key, JSON.stringify(cacheData));
      } catch (e) {
        console.warn('LocalStorage full, clearing old cache');
        this.clearOldCache();
      }
    }
  }
  
  get(key, type = 'orderList') {
    const config = this.cacheConfig[type];
    let cacheData;
    
    if (config.storage === 'memory') {
      cacheData = this.memoryCache.get(key);
    } else {
      const stored = this.storageCache.getItem(key);
      cacheData = stored ? JSON.parse(stored) : null;
    }
    
    if (!cacheData) return null;
    
    // 检查是否过期
    if (Date.now() - cacheData.timestamp > cacheData.ttl) {
      this.delete(key, type);
      return null;
    }
    
    return cacheData.value;
  }
  
  delete(key, type = 'orderList') {
    const config = this.cacheConfig[type];
    
    if (config.storage === 'memory') {
      this.memoryCache.delete(key);
    } else {
      this.storageCache.removeItem(key);
    }
  }
  
  clearOldCache() {
    const now = Date.now();
    for (let i = 0; i < this.storageCache.length; i++) {
      const key = this.storageCache.key(i);
      try {
        const data = JSON.parse(this.storageCache.getItem(key));
        if (data.timestamp && now - data.timestamp > data.ttl) {
          this.storageCache.removeItem(key);
        }
      } catch (e) {
        // 忽略非JSON数据
      }
    }
  }
  
  clear(type) {
    if (type) {
      const config = this.cacheConfig[type];
      if (config.storage === 'memory') {
        this.memoryCache.clear();
      } else {
        Object.keys(this.storageCache)
          .filter(key => key.startsWith(type))
          .forEach(key => this.storageCache.removeItem(key));
      }
    } else {
      this.memoryCache.clear();
      this.storageCache.clear();
    }
  }
}

// 使用缓存的Hook
const useCachedData = (key, fetcher, type = 'orderList') => {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const cache = useMemo(() => new CacheManager(), []);
  
  const fetchData = async (forceRefresh = false) => {
    // 尝试从缓存获取
    if (!forceRefresh) {
      const cached = cache.get(key, type);
      if (cached) {
        setData(cached);
        return cached;
      }
    }
    
    // 从服务器获取
    setLoading(true);
    try {
      const result = await fetcher();
      cache.set(key, result, type);
      setData(result);
      return result;
    } finally {
      setLoading(false);
    }
  };
  
  useEffect(() => {
    fetchData();
  }, [key]);
  
  return { data, loading, refresh: () => fetchData(true) };
};
```

```javascript
// 前端筛选Hook
const useOrderFilter = (orders) => {
  const [filters, setFilters] = useState({
    keyword: '',
    dateRange: { start: null, end: null },
    priceRange: { min: 0, max: Infinity },
    warehouses: [],
    countries: [],
    statuses: [],
    sortBy: 'created_time',
    sortOrder: 'desc'
  });
  
  const filteredOrders = useMemo(() => {
    let result = [...orders];
    
    // 关键词筛选
    if (filters.keyword) {
      const keyword = filters.keyword.toLowerCase();
      result = result.filter(order => 
        order.express_num?.toLowerCase().includes(keyword) ||
        order.order_sn?.toLowerCase().includes(keyword) ||
        order.class_name?.toLowerCase().includes(keyword)
      );
    }
    
    // 日期范围筛选
    if (filters.dateRange.start && filters.dateRange.end) {
      result = result.filter(order => {
        const orderDate = new Date(order.created_time);
        return orderDate >= filters.dateRange.start && 
               orderDate <= filters.dateRange.end;
      });
    }
    
    // 价格范围筛选
    result = result.filter(order => 
      order.price >= filters.priceRange.min && 
      order.price <= filters.priceRange.max
    );
    
    // 仓库筛选
    if (filters.warehouses.length > 0) {
      result = result.filter(order => 
        filters.warehouses.includes(order.storage_id)
      );
    }
    
    // 国家筛选
    if (filters.countries.length > 0) {
      result = result.filter(order => 
        filters.countries.includes(order.country_id)
      );
    }
    
    // 状态筛选
    if (filters.statuses.length > 0) {
      result = result.filter(order => 
        filters.statuses.includes(order.status)
      );
    }
    
    // 排序
    result.sort((a, b) => {
      const aVal = a[filters.sortBy];
      const bVal = b[filters.sortBy];
      
      if (filters.sortOrder === 'asc') {
        return aVal > bVal ? 1 : -1;
      } else {
        return aVal < bVal ? 1 : -1;
      }
    });
    
    return result;
  }, [orders, filters]);
  
  return { filteredOrders, filters, setFilters };
};
```

```javascript
// 下拉刷新组件
const PullToRefresh = ({ onRefresh, children }) => {
  const [pulling, setPulling] = useState(false);
  const [refreshing, setRefreshing] = useState(false);
  const [pullDistance, setPullDistance] = useState(0);
  const containerRef = useRef(null);
  const startY = useRef(0);
  
  const threshold = 80;
  
  const handleTouchStart = (e) => {
    if (containerRef.current.scrollTop === 0) {
      setPulling(true);
      startY.current = e.touches[0].clientY;
    }
  };
  
  const handleTouchMove = (e) => {
    if (!pulling) return;
    
    const currentY = e.touches[0].clientY;
    const distance = Math.max(0, currentY - startY.current);
    
    // 阻尼效果
    const damping = 0.5;
    const dampedDistance = distance * damping;
    setPullDistance(dampedDistance);
    
    // 防止页面滚动
    if (distance > 0) {
      e.preventDefault();
    }
  };
  
  const handleTouchEnd = async () => {
    if (pullDistance > threshold && !refreshing) {
      setRefreshing(true);
      try {
        await onRefresh();
      } finally {
        setRefreshing(false);
      }
    }
    
    setPulling(false);
    setPullDistance(0);
  };
  
  return (
    <div 
      ref={containerRef}
      className="h-full overflow-auto"
      onTouchStart={handleTouchStart}
      onTouchMove={handleTouchMove}
      onTouchEnd={handleTouchEnd}
    >
      {/* 下拉指示器 */}
      <div 
        className="flex items-center justify-center transition-all duration-200"
        style={{ 
          height: `${pullDistance}px`,
          opacity: Math.min(pullDistance / threshold, 1)
        }}
      >
        {refreshing ? (
          <div className="flex items-center gap-2 text-blue-600">
            <div className="w-5 h-5 border-2 border-blue-600 border-t-transparent rounded-full animate-spin" />
            <span className="text-sm font-medium">刷新中...</span>
          </div>
        ) : (
          <div className="flex items-center gap-2 text-gray-600">
            <svg 
              className="w-5 h-5 transition-transform"
              style={{ transform: `rotate(${Math.min(pullDistance / threshold, 1) * 180}deg)` }}
              fill="none" 
              stroke="currentColor" 
              viewBox="0 0 24 24"
            >
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M19 14l-7 7m0 0l-7-7m7 7V3" />
            </svg>
            <span className="text-sm font-medium">
              {pullDistance > threshold ? '释放刷新' : '下拉刷新'}
            </span>
          </div>
        )}
      </div>
      
      {/* 内容 */}
      <div style={{ transform: `translateY(${pullDistance}px)`, transition: pulling ? 'none' : 'transform 0.3s' }}>
        {children}
      </div>
    </div>
  );
};

// 上拉加载组件
const InfiniteScroll = ({ onLoadMore, hasMore, loading, children }) => {
  const observerRef = useRef(null);
  const sentinelRef = useRef(null);
  
  useEffect(() => {
    observerRef.current = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && hasMore && !loading) {
          onLoadMore();
        }
      },
      { threshold: 0.1 }
    );
    
    if (sentinelRef.current) {
      observerRef.current.observe(sentinelRef.current);
    }
    
    return () => {
      if (observerRef.current) {
        observerRef.current.disconnect();
      }
    };
  }, [hasMore, loading, onLoadMore]);
  
  return (
    <div className="space-y-4">
      {children}
      
      <div ref={sentinelRef} className="py-4">
        {loading && (
          <div className="flex items-center justify-center gap-2 text-gray-500">
            <div className="w-5 h-5 border-2 border-gray-400 border-t-transparent rounded-full animate-spin" />
            <span className="text-sm">加载中...</span>
          </div>
        )}
        
        {!hasMore && !loading && (
          <div className="text-center text-gray-400 text-sm">
            没有更多数据了
          </div>
        )}
      </div>
    </div>
  );
};
```

```javascript
      totalAmount,
      totalWeight,
      avgPrice,
      avgWeight,
      statusDistribution,
      warehouseDistribution,
      countryDistribution
    };
  }, [orders]);
  
  return statistics;
```

```javascript
// 搜索历史管理
const useSearchHistory = (maxHistory = 10) => {
  const [history, setHistory] = useState(() => {
    const saved = localStorage.getItem('search_history');
    return saved ? JSON.parse(saved) : [];
  });
  
  const addHistory = (keyword) => {
    if (!keyword || keyword.trim() === '') return;
    
    const newHistory = [
      keyword,
      ...history.filter(item => item !== keyword)
    ].slice(0, maxHistory);
    
    setHistory(newHistory);
    localStorage.setItem('search_history', JSON.stringify(newHistory));
  };
  
  const removeHistory = (keyword) => {
    const newHistory = history.filter(item => item !== keyword);
    setHistory(newHistory);
    localStorage.setItem('search_history', JSON.stringify(newHistory));
  };
  
  const clearHistory = () => {
    setHistory([]);
    localStorage.removeItem('search_history');
  };
  
  return { history, addHistory, removeHistory, clearHistory };
};

// 搜索组件
const SearchWithHistory = ({ onSearch }) => {
  const [keyword, setKeyword] = useState('');

  const handleSearch = (searchKeyword) => {
    const finalKeyword = searchKeyword || keyword;

      onSearch(finalKeyword);

    }

  return (
    <div className="relative">

            >
              清空
            </button>
          </div>
          
          <div className="p-2">
            {history.map((item, index) => (
```
